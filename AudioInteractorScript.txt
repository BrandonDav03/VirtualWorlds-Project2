const listener = new THREE.AudioListener();
camera.add( listener );
const userSound = new THREE.Audio( listener );
const globalAudioLoader = new THREE.AudioLoader();
let activated = false;

function activateSound(soundLink) {

    globalAudioLoader.load(
        soundLink,
        function (buffer) {
            userSound.stop();
            userSound.setBuffer(buffer);
            userSound.setLoop(false);
            userSound.setVolume(0.5);
            userSound.play();
            //userSound.stop(8);
        }
    );
}


let roundActive = false;
let timeRemaining = 0;
let correctChordIndex = -1;
let game = -1;

function playGame(soundArray, buttonArray) {
    if (roundActive) return;
    if(timeRemaining < 0) return;
    if (soundArray.length !== buttonArray.length) return;
    if (soundArray.length === 0) return;

    // Choose the chord the player must identify.
    correctChordIndex = Math.floor(Math.random() * soundArray.length);
    roundActive = true;
    timeRemaining = 10;

    console.log("Listen to the chord!");

    // Play the randomly selected chord.
    // See the audio note below about resetting `activated`.
    activateSound(soundArray[correctChordIndex]);

    // Lose if the player doesn't answer within 10 seconds.
}

function finishRound(won, message) {
    if (!roundActive) return;

    let gameText;

    switch(game) {
        case 0:
           gameText = chordText;
           break;
        case 1:
           gameText = scaleText;
           break;
        case 2:
           gameText = intervalText;
           break;
        default:
           gameText = welcomeText;
           break;
    }

    roundActive = false;
    timeRemaining = -1;

    if (won) {
        gameText.text = message;
    } else {
        gameText.text = message;
    }
}

function submitAnswer(buttonIndex) {
    // Ignore clicks when there isn't an active round.
    if (!roundActive) return;

    if (buttonIndex === correctChordIndex) {
        finishRound(true, "Correct!");
    } else {
        finishRound(false, "Incorrect!");
    }
}
